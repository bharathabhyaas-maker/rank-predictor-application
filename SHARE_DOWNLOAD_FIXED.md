```typescript
const handleShare = async () => {
  if (predictionData) {
    const shareText = `My Rank Prediction for ${predictionData.examName}:
📊 Predicted Percentile: ${predictionData.percentile.predictedPercentile.toFixed(1)}%
🎯 Expected Rank: ${predictionData.rankRange.predictedRank.toLocaleString()}
📈 Rank Range: ${predictionData.rankRange.minRank.toLocaleString()} - ${predictionData.rankRange.maxRank.toLocaleString()}
💯 Score: ${predictionData.totalScore}
🔮 Prediction Method: ${predictionData.calculationMethod}

Check your rank prediction at: ${window.location.href}`

    try {
      if (navigator.share) {
        // Use Web Share API if available
        await navigator.share({
          title: `Rank Prediction - ${predictionData.examName}`,
          text: shareText,
          url: window.location.href
        })
      } else {
        // Fallback: Copy to clipboard
        await navigator.clipboard.writeText(shareText)
        alert('Results copied to clipboard!')
      }
    } catch (error) {
      console.error('Error sharing:', error)
      alert('Failed to share results')
    }
  }
}
```

#### **2. Added Download Functionality**
```typescript
const handleDownload = () => {
  if (predictionData) {
    const downloadText = `RANK PREDICTION RESULTS
========================

Exam: ${predictionData.examName}
Date: ${new Date().toLocaleDateString()}

PREDICTION DETAILS
------------------
Your Score: ${predictionData.totalScore}
Predicted Percentile: ${predictionData.percentile.predictedPercentile.toFixed(1)}%
Expected Rank: ${predictionData.rankRange.predictedRank.toLocaleString()}

RANK RANGE
----------
Best Case: ${predictionData.rankRange.minRank.toLocaleString()} (Percentile: ${predictionData.percentile.minPercentile.toFixed(1)}%)
Expected: ${predictionData.rankRange.predictedRank.toLocaleString()} (Percentile: ${predictionData.percentile.predictedPercentile.toFixed(1)}%)
Worst Case: ${predictionData.rankRange.maxRank.toLocaleString()} (Percentile: ${predictionData.percentile.maxPercentile.toFixed(1)}%)

PREDICTION METHOD
-----------------
${predictionData.calculationMethod}

Generated by RankPredict
${window.location.href}`

    // Create and download file
    const blob = new Blob([downloadText], { type: 'text/plain' })
    const url = window.URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `rank-prediction-${predictionData.examName.replace(/\s+/g, '-').toLowerCase()}-${new Date().toISOString().split('T')[0]}.txt`
    document.body.appendChild(a)
    a.click()
    document.body.removeChild(a)
    window.URL.revokeObjectURL(url)
  }
}
```
